% 根据学位类型设置进展报告页眉。
% 双面模式在偶数页显示学校与进展报告类型，奇数页保留章节标题。
\ExplSyntaxOn
\NewDocumentCommand{\MidtermRunningHeader}{}
  {
    西北农林科技大学
    \int_compare:nNnTF {\g__nwafu_thesis_type_int} = {1}
      {博士进展报告} {硕士进展报告}
  }
\cs_if_exist:NT \__nwafu_header_even:
  {
    \cs_set:Npn \__nwafu_header_even:
      {\small \nouppercase{\MidtermRunningHeader}}
  }
\cs_if_exist:NT \__nwafu_header_center:
  {
    \cs_set:Npn \__nwafu_header_center:
      {\small \nouppercase{\MidtermRunningHeader}}
  }
\NewDocumentCommand{\MidtermApplyRunningHeader}{}
  {
    \bool_if:NTF \g__nwafu_twoside_bool
      {\nwafuhead[EC]{\small\nouppercase{\MidtermRunningHeader}}}
      {\nwafuhead[C]{\small\nouppercase{\MidtermRunningHeader}}}
  }
\ExplSyntaxOff

% frontmatter is called inside nwafuthesis's \makefront.
% Apply the midterm header immediately after frontmatter configures page styles.
\AddToHook{cmd/frontmatter/after}{\MidtermApplyRunningHeader}
